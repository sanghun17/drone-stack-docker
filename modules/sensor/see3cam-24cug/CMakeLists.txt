cmake_minimum_required(VERSION 3.10)
project(see3cam_demand_publisher)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs std_msgs camera_info_manager)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d imgcodecs)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED libavcodec libavutil libswscale)
add_executable(see3cam_node see3cam_node.cpp vendor/usb_cam/usb_cam.cpp)
target_compile_features(see3cam_node PRIVATE cxx_std_14)
target_include_directories(see3cam_node BEFORE PRIVATE vendor/usb_cam/include)
target_include_directories(see3cam_node PRIVATE ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${FFMPEG_INCLUDE_DIRS})
target_link_libraries(see3cam_node ${catkin_LIBRARIES} ${OpenCV_LIBRARIES} ${FFMPEG_LIBRARIES})
target_compile_options(see3cam_node PRIVATE -Wall -Wextra)
